cmake_minimum_required(VERSION 3.18)
project(rwkv_interp LANGUAGES CXX)
set(LLAMA_BUILD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../build" CACHE PATH "Existing llama.cpp build")
get_filename_component(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_library(LLAMA_LIBRARY NAMES llama PATHS "${LLAMA_BUILD_DIR}/bin" REQUIRED NO_DEFAULT_PATH)
find_library(GGML_LIBRARY NAMES ggml PATHS "${LLAMA_BUILD_DIR}/bin" REQUIRED NO_DEFAULT_PATH)
find_library(GGML_BASE_LIBRARY NAMES ggml-base PATHS "${LLAMA_BUILD_DIR}/bin" REQUIRED NO_DEFAULT_PATH)
add_executable(rwkv-interp-prove prove.cpp)
add_executable(rwkv-interp-collect collect.cpp)
set(interp_targets rwkv-interp-prove rwkv-interp-collect)
foreach(experiment layer-swap read-mediation reusable-query read-linearity)
    add_executable(rwkv-interp-${experiment} read-query/${experiment}.cpp)
    list(APPEND interp_targets rwkv-interp-${experiment})
endforeach()
foreach(target IN LISTS interp_targets)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE RWKV_INTERP_BACKEND_DIR="${LLAMA_BUILD_DIR}/bin")
    target_include_directories(${target} PRIVATE "${LLAMA_SOURCE_DIR}/include" "${LLAMA_SOURCE_DIR}/src" "${LLAMA_SOURCE_DIR}/ggml/include")
    target_link_libraries(${target} PRIVATE "${LLAMA_LIBRARY}" "${GGML_LIBRARY}" "${GGML_BASE_LIBRARY}")
endforeach()
